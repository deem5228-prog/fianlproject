Python import io # อ่านข้อมูลไบนารี (รูปภาพจาก bytes) import os # จัดการไฟล์/โฟลเดอร์ import cv2 # OpenCV: ประมวลผลภาพ ตรวจจับไข่แดง import numpy as np # จัดการตัวเลข/อาเรย์พิกเซล from PIL import Image # จัดการรูปภาพ (เปิด/แปลง) from skimage.color import rgb2lab # แปลงสี RGB → CIELAB

1.1 คลาส ColorFeatureList — กล่องเก็บค่าสีอัจฉริยะ

คลาสนี้เก็บค่าสี 6 ค่า: [R, G, B, L*, a*, b*] ในรูปแบบ List ธรรมดา แต่เพิ่มความสามารถพิเศษคือ เรียกค่าได้ด้วยชื่อ เหมือน Dictionary

Python class ColorFeatureList(list): # แมปชื่อ → ตำแหน่ง index ในลิสต์ _KEY_MAP = { 'r': 0, 'g': 1, 'b': 2, 'l': 3, 'a': 4, 'b_lab': 5, } def __getitem__(self, item): if isinstance(item, str): return super().__getitem__(self._KEY_MAP[item]) return super().__getitem__(item)
💡 วิธีใช้งาน
feats = ColorFeatureList([226, 131, 23, 64.2, 29.8, 65.9])
เรียกด้วย index: feats[0] → 226 (ค่า R)
เรียกด้วยชื่อ: feats['r'] → 226 (ค่า R)
เรียกด้วยชื่อ: feats['b_lab'] → 65.9 (ค่า b* ในระบบสี CIELAB)

1.2 ฟังก์ชัน auto_crop_yolk() — ครอปไข่แดงด้วย OpenCV

ฟังก์ชันนี้ทำหน้าที่ ตรวจจับตำแหน่งไข่แดง ในรูปภาพ แล้วตัดเฉพาะส่วนไข่แดงออกมาโดยอัตโนมัติ

🖼️รับภาพBGR/RGB/PIL/bytes
➜
🌈แปลง → HSVBGR → HSV
➜
🎭สร้าง Maskกรองสีเหลือง-ส้ม
➜
🧹ทำความสะอาดMorphology
➜
✂️ครอปวงกลม → สี่เหลี่ยม

ขั้นตอนที่ 1: แปลงสี BGR → HSV

Python hsv = cv2.cvtColor(bgr, cv2.COLOR_BGR2HSV)

ภาพที่ OpenCV อ่านมาจะอยู่ในรูปแบบ BGR (น้ำเงิน-เขียว-แดง) เราต้องแปลงเป็น HSV (Hue-Saturation-Value) เพราะง่ายต่อการกรองสีตามช่วงสีที่ต้องการ

🌈 HSV คืออะไร?
H (Hue) = เฉดสี (0-180 ในระบบ OpenCV) เช่น แดง=0, เขียว=60, น้ำเงิน=120
S (Saturation) = ความเข้มสี (0=สีซีด, 255=สีจัด)
V (Value) = ความสว่าง (0=มืด, 255=สว่าง)
ไข่แดงจะมีสีในช่วง เหลือง-ส้ม → H อยู่ประมาณ 10-28 (ในสเกล OpenCV)

ขั้นตอนที่ 2: สร้าง Mask กรองเฉพาะสีไข่แดง

Python # กำหนดขอบเขตล่างและบนของสีไข่แดง lower_hsv = np.array([10, 80, 80]) # Hue 10, Sat 80, Val 80 upper_hsv = np.array([28, 255, 250]) # Hue 28, Sat 255, Val 250 mask = cv2.inRange(hsv, lower_hsv, upper_hsv)

cv2.inRange() จะสร้าง "หน้ากาก" (Mask) ที่เป็นภาพขาว-ดำ: สีขาว = ไข่แดง, สีดำ = ไม่ใช่

ขั้นตอนที่ 3: ทำความสะอาด Mask (Morphological Operations)

Python kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (7, 7)) mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, kernel) # อุดรูเล็กๆ mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, kernel) # ลบจุดรบกวน

Close = อุดรูเล็กๆ ที่อยู่ภายในไข่แดง, Open = ลบจุดเล็กๆ ที่ไม่ใช่ไข่แดง (เช่น สะท้อนแสง)

ขั้นตอนที่ 4: หาวงกลมที่ล้อมไข่แดง แล้วครอปออกมา

Python # หาเส้นขอบ (Contour) ของไข่แดง contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE) largest_contour = max(contours, key=cv2.contourArea) # หาวงกลมที่เล็กที่สุดที่ครอบไข่แดงได้พอดี (cx, cy), radius = cv2.minEnclosingCircle(largest_contour) # ตัดภาพเป็นสี่เหลี่ยมจตุรัส (ขยาย 8% จากรัศมี) crop_size = radius * 1.08

1.3 ฟังก์ชัน extract_mean_rgb() — วัดค่าสีเฉลี่ย (Center Circular Mask 42%)

ฟังก์ชันนี้คำนวณ ค่าสีเฉลี่ย R, G, B โดยใช้เฉพาะ พิกเซลตรงกลางภาพ ที่อยู่ในวงกลมรัศมี 42% ของภาพเท่านั้น

⭕ ทำไมต้องใช้วงกลมแค่ 42%?
เพราะภาพไข่แดงที่ครอปมาแล้ว ขอบๆ ภาพอาจมีไข่ขาวหรือพื้นหลังปนอยู่ การใช้ Circular Mask จะทำให้เราอ่านสีเฉพาะ "เนื้อไข่แดงบริสุทธิ์" ตรงกลาง ไม่มีสีอื่นมาปนเปื้อน
Python # กำหนดจุดศูนย์กลางและรัศมีวงกลม cy, cx = h // 2, w // 2 radius = int(min(h, w) * 0.42) # 42% ของด้านที่สั้นกว่า # สร้าง Mask วงกลม: True = อยู่ในวงกลม, False = อยู่นอก y_coords, x_coords = np.ogrid[:h, :w] mask = (x_coords - cx)**2 + (y_coords - cy)**2 <= radius**2 # คำนวณค่าเฉลี่ย R, G, B เฉพาะพิกเซลในวงกลม mean_r = float(np.mean(np_img[:, :, 0][mask])) mean_g = float(np.mean(np_img[:, :, 1][mask])) mean_b = float(np.mean(np_img[:, :, 2][mask]))

สมการ (x - cx)² + (y - cy)² ≤ r² คือ สมการวงกลม พื้นฐานที่ตรวจว่าพิกเซลนั้นอยู่ในวงกลมหรือไม่

1.4 ฟังก์ชัน rgb_to_cielab() — แปลง RGB เป็น CIELAB

🔬 CIELAB คืออะไร?
เป็นระบบสีมาตรฐานที่ออกแบบมาให้ ตรงกับการมองเห็นของมนุษย์:
L* = ความสว่าง (0=ดำ, 100=ขาว)
a* = แกนเขียว↔แดง (ค่าลบ=เขียว, ค่าบวก=แดง) — ค่านี้สำคัญมากสำหรับไข่แดง!
b* = แกนน้ำเงิน↔เหลือง (ค่าลบ=น้ำเงิน, ค่าบวก=เหลือง)
Python def rgb_to_cielab(r, g, b): # ปรับค่า 0-255 ให้เป็น 0-1 rgb_norm = np.array([[[r / 255.0, g / 255.0, b / 255.0]]]) # ใช้ scikit-image แปลงตามมาตรฐาน CIE D65 lab_arr = rgb2lab(rgb_norm)[0, 0] return {'l': lab_arr[0], 'a': lab_arr[1], 'b': lab_arr[2]}

1.5 ฟังก์ชัน extract_color_features() — รวมร่างทุกอย่าง

ฟังก์ชันตัวหลักที่ ร้อยเรียงทุกอย่างเข้าด้วยกัน: รับภาพ → วัด RGB → แปลง CIELAB → คืนค่า 6 ตัว

Python def extract_color_features(image_input): rgb = extract_mean_rgb(image_input) # ได้ R, G, B lab = rgb_to_cielab(rgb['r'], rgb['g'], rgb['b']) # ได้ L*, a*, b* return ColorFeatureList([ rgb['r'], rgb['g'], rgb['b'], lab['l'], lab['a'], lab['b'] ]) # ผลลัพธ์: [226.0, 131.0, 23.0, 64.2, 29.8, 65.9]

Python # ตัดภาพเป็นสี่เหลี่ยมจัตุรัส โดยกึ่งกลางที่ centroid ของไข่แดง box_size = int(max(bbox_w, bbox_h) * (1.0 + padding_ratio)) cropped_img = img.crop((x1, y1, x1 + actual_size, y1 + actual_size)) # บันทึก Bounding Box ในรูปแบบ YOLO (เผื่อใช้ Deep Learning ในอนาคต) # "0 0.512 0.498 0.342 0.356" → คลาส 0 (ไข่แดง), ตำแหน่ง x,y,w,h แบบ normalize

4.2 Stratified 5-Fold Cross-Validation คืออะไร?

🔄 Cross-Validation แบบ Stratified

แทนที่จะแบ่งข้อมูลเป็น "ชุดเทรน" กับ "ชุดทดสอบ" แค่ครั้งเดียว เราแบ่งข้อมูลเป็น 5 ส่วน (Fold) แล้ววนทดลอง 5 รอบ:

รอบ 1: ใช้ส่วน 1 ทดสอบ, ส่วน 2-5 เทรน
รอบ 2: ใช้ส่วน 2 ทดสอบ, ส่วน 1,3-5 เทรน
... และต่อไปจนครบ 5 รอบ

คำว่า Stratified หมายถึงทุก Fold จะมีสัดส่วนของแต่ละคลาสเท่าๆ กัน (เช่น class4 มี 50 รูป ก็จะถูกแบ่งเป็น 10 รูปต่อ Fold) ทำให้ผลลัพธ์น่าเชื่อถือมากขึ้น

Python # ตั้งค่า Cross-Validation skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42) for name, pipeline in models.items(): for train_idx, val_idx in skf.split(X, y): X_tr, y_tr = X[train_idx], y[train_idx] # ข้อมูลเทรน (80%) X_te, y_te = X[val_idx], y[val_idx] # ข้อมูลทดสอบ (20%) pipeline.fit(X_tr, y_tr) # สอนโมเดล p_te = pipeline.predict(X_te) # ให้โมเดลทำนาย # วัดผล te_r2.append(r2_score(y_te, p_te)) # R² = ความแม่นยำรวม te_mae.append(mean_absolute_error(y_te, p_te)) # MAE = ค่าผิดเฉลี่ย

5.3 Sanity Check — พิสูจน์ว่าสมการคณิตศาสตร์ให้ผลเหมือนกัน

ก่อนจบ สคริปต์จะทดสอบว่า การคำนวณด้วยคณิตศาสตร์ล้วนๆ ให้คำตอบเดียวกันกับ scikit-learn:

สูตร SVR (RBF Kernel) ที่ฝั่งมือถือใช้คำนวณ:
f(x) = Σ αᵢ · exp(-γ · ‖x - svᵢ‖²) + b
αᵢ = dual_coef[i], γ = gamma, svᵢ = support_vectors[i], b = intercept
Python # จำลองสมการ SVR ด้วยเลขบวกลบคูณหารล้วนๆ scaled_x = (sample - scaler_mean) / scaler_scale # ปรับสเกล dists = np.sum((sv_arr - scaled_x)**2, axis=1) # ระยะห่างยกกำลัง 2 k_vals = np.exp(-gamma * dists) # RBF Kernel result = np.sum(coef_arr * k_vals) + intercept # รวมผล # ผลลัพธ์: # Scikit-Learn Output: 10.234567 # Pure Math Output: 10.234567 # Absolute Difference: 0.0e+00 (ตรงกัน 100%!)

📐 Epsilon Tube ของ SVR คืออะไร?

SVR จะพยายามหาเส้นทำนาย f(x) ที่มี "ท่อ" (Tube) ล้อมรอบ ±ε (ε = 0.1 ในโปรเจกต์นี้)

  • จุดข้อมูลที่อยู่ ภายในท่อ → ไม่ถูกลงโทษ (SVR ถือว่าทำนายถูก)
  • จุดข้อมูลที่อยู่ นอกท่อ → ถูกลงโทษตามสัดส่วนระยะห่าง
  • Support Vectors = จุดที่อยู่บนขอบท่อหรือนอกท่อ — เป็นจุดที่ SVR "จำ" ไว้เพื่อกำหนดรูปร่างของเส้นทำนาย
Python # ใช้ฟีเจอร์ a* (Redness) แกนเดียว เพื่อวาดเป็น 2D ให้เข้าใจง่าย X_a = df[['a']].values # แกน X = ค่าความแดง y = df['fan_score'].values # แกน Y = คะแนน Fan Score # วาดส่วนประกอบ 4 ชั้น: # 1. แถบสีเหลืองอ่อน = Epsilon Tube (±0.1) plt.fill_between(x, y_pred - eps, y_pred + eps, color='#FDE68A') # 2. เส้นน้ำเงิน = เส้นทำนาย SVR plt.plot(x, y_pred, color='#1D4ED8', linewidth=3) # 3. จุดส้ม = ตัวอย่างไข่แดงทั้งหมด plt.scatter(X_a, y, color='#F97316') # 4. จุดเขียว = Support Vectors plt.scatter(sv_x, sv_y, color='#10B981')
